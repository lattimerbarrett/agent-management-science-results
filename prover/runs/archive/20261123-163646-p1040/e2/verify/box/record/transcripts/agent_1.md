import sympy as S

T, u, a, b, t, s, z = S.symbols("T u a b t s z")
I = S.eye(2)

def lie(X, Y):
    return X*Y - Y*X

def comm(X, Y):
    return (X*Y*X.inv()*Y.inv()).applyfunc(S.cancel)

def equal(X, Y):
    return all(S.cancel(v) == 0 for v in X-Y)

def x(a):
    return S.Matrix([[1, a], [0, 1]])

def y(b):
    return S.Matrix([[1, 0], [b, 1]])

def h(t):
    return S.diag(t, 1/t)

# Derivations fixing Y are Z[T]-linear, where T = ad(Y).
D = S.diag(T, 0)
A = S.Matrix([[0, 0], [T, 0]])
Bu = S.Matrix([[0, -u], [0, 0]])

# These are identities in independent indeterminates T,u.
assert equal(lie(D, Bu), Bu.subs(u, T*u))
assert equal(lie(D, lie(A, Bu)), S.zeros(2))

# Exact root recursion.
assert equal(comm(S.diag(t, 1), x(b)), x((t-1)*b))

# Exact opposite-root Gaussian factorization.
delta = 1-a*b
assert equal(
    comm(x(a), y(b)),
    x(-a*a*b/delta) * h(1/delta) * y(a*b*b/delta)
)

# Actual column-convention witness matrices.
for c in range(3, 17):
    Dg = S.diag(1+s, 1)
    B = x(-s)
    for _ in range(c-3):
        B = comm(Dg, B)

    assert equal(B, x(-s**(c-2)))
    W = comm(Dg, comm(y(s), B))

    valuations = []
    for entry in W-I:
        num, den = S.fraction(S.cancel(entry))
        assert den.subs(s, 0) != 0
        if num != 0:
            valuations.append(min(
                monomial[0]
                for monomial, coefficient in S.Poly(num, s).terms()
            ))
    assert min(valuations) == c+1

# Eisenstein field choices and their exact ramification thresholds.
for c in range(3, 65):
    odd_primes = [p for p in S.factorint(c) if p != 2]
    if odd_primes:
        p = min(odd_primes)
        q = c // p
        f = S.Poly(sum((1+z**q)**j for j in range(p)), z)
        e = (p-1)*q
    else:
        p = 2
        q = c // 4
        f = S.Poly(z**(2*q)-2*z**q+2, z)
        e = 2*q

    coefficients = f.all_coeffs()
    assert f.degree() == e
    assert coefficients[0] == 1
    assert all(int(v) % p == 0 for v in coefficients[1:])
    assert int(coefficients[-1]) % (p*p) != 0
    assert p*e == c*(p-1)

# Independent exact norm checks in the basic odd-prime cases.
# Local cyclotomic reciprocity makes norm != 1 mod p^2
# an independent check of the nontrivial critical Hilbert symbol.
for p in [3, 5, 7, 11, 13, 17, 19]:
    cyclotomic = S.cyclotomic_poly(p, z)
    norm = int(S.resultant(
        cyclotomic, 1+(z-1)**(p-1), z
    ))
    assert norm % (p*p) == 1+p

print("All exact checks passed.")
